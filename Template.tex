\section{LaTeX Beamer 基本说明}

\begin{frame}[fragile]{基本用法}
    最基本的 frame 环境结构如下，一个 frame 就是一张幻灯片：
    
    \begin{code}[TeX]
        \begin{frame}[<参数选项>]{幻灯片主标题}{幻灯片副标题}
            这里是幻灯片的正文内容。
        \end|\empty|{frame}
    \end{code}

    可以使用 block 环境把公式输进去：

    \begin{block}{block环境}
        \begin{equation}
            \alpha , \beta
        \end{equation}
    \end{block}

\end{frame}

\begin{frame}
    还用 alertblock 环境强调：

    \begin{alertblock}{alertblock环境}
        要强调的东西
    \end{alertblock}
\end{frame}

% ---------------- 第二部分 ----------------
\section{图片插入}

\begin{frame}{单张图片}
    \begin{figure}[htbp]
	    \centering
	    \includegraphics[width=0.5\linewidth]{Figures/时谐电流源J.pdf}
	    \caption{图名\label{图名}}
    \end{figure}
\end{frame}

\begin{frame}{多个子图(以1×2子图为例)}
    \begin{figure}[H]
        \centering
        % 第一行
        \begin{subfigure}[b]{0.45\textwidth}%[b]表示这一行所有子图的底部边缘(通常是子图标题的最后一行水平基线)对齐在同一条水平线上。这是学术排版中最常用、看起来最整齐的设置。
            \centering
            \includegraphics[width=\textwidth]{Figures/时谐电流源J.pdf}
            \caption{SubFigure1}
            \label{fig:sub1}
        \end{subfigure}
        \hfill % 使用 hfill 自动推开间距，比手动 hspace 更好控制
        \begin{subfigure}[b]{0.45\textwidth}
            \centering
            \includegraphics[width=\textwidth]{Figures/时谐电流源J.pdf}
            \caption{SubFigure2}
            \label{fig:sub2}
        \end{subfigure}
    
        \caption{1×2布局\label{fig:main}}
    \end{figure}
\end{frame}
